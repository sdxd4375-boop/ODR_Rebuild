```python
"""Environment self-check: catch a broken venv before it wastes an hour."""
import importlib, os, sys

REQUIRED = ["aiohttp", "fastapi", "sqlalchemy", "asyncpg", "alembic",
            "langgraph", "langchain_core", "numpy", "matplotlib", "pptx", "docx"]

def main() -> int:
    print(f"python  : {sys.version.split()[0]} ({sys.executable})")
    if sys.version_info < (3, 11):
        print("FAIL: requires Python >= 3.11")
        return 1
    bad = []
    for name in REQUIRED:
        try:
            importlib.import_module(name)
            print(f"ok      : {name}")
        except Exception as exc:
            bad.append((name, exc))
            print(f"FAIL    : {name} -> {exc}")
    for path in ("src/open_deep_research", "src/server", "tests/server"):
        print(f"{'ok' if os.path.isdir(path) else 'FAIL'}      : {path}")
    return 1 if bad else 0

if __name__ == "__main__":
    raise SystemExit(main())
```

`health.py`：

```python
from fastapi import APIRouter, Request
from fastapi.responses import JSONResponse

from server.deps import auth_mode

router = APIRouter()


@router.get("/health")
async def health(request: Request):
    """Liveness plus readiness; 503 when a required subsystem is unavailable."""
    db_ready = bool(getattr(request.app.state, "db_ready", False))
    graph_ready = bool(getattr(request.app.state, "graph_ready", False))
    ok = db_ready and graph_ready
    body = {
        "status": "ok" if ok else "degraded",
        "db": db_ready,
        "graph": graph_ready,
        "auth_mode": auth_mode(),
        "detail": "" if ok else "database or graph unavailable (check DATABASE_URL and startup logs)",
    }
    return JSONResponse(body, status_code=200 if ok else 503)
```

```python
async def _tokens_used_today(user_id: str) -> int:
    """Sum of total_tokens for this user since UTC midnight (0 when DB absent)."""
    from datetime import datetime, timezone

    from sqlalchemy import func, select

    from server.db import get_session_factory
    from server.models import UsageEvent

    try:
        factory = get_session_factory()
    except RuntimeError:
        return 0
    midnight = datetime.now(timezone.utc).replace(hour=0, minute=0, second=0, microsecond=0)
    async with factory() as db:
        total = await db.scalar(
            select(func.coalesce(func.sum(UsageEvent.total_tokens), 0)).where(
                UsageEvent.user_id == user_id, UsageEvent.created_at >= midnight
            )
        )
    return int(total or 0)


async def enforce_token_quota(user_id: str) -> None:
```

```python
FORBIDDEN_CALLS = {"eval", "exec", "compile", "__import__", "open", "input",
                   "globals", "locals", "vars", "getattr", "setattr", "delattr",
                   "breakpoint", "memoryview", "exit", "quit"}
FORBIDDEN_ATTR_PREFIX = "__"


def check_forbidden_nodes(code: str) -> list[str]:
    """Return forbidden call names / dunder attribute accesses found in the AST."""
    tree = ast.parse(code)
    found: list[str] = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Call) and isinstance(node.func, ast.Name):
            if node.func.id in FORBIDDEN_CALLS:
                found.append(node.func.id)
        elif isinstance(node, ast.Attribute) and node.attr.startswith(FORBIDDEN_ATTR_PREFIX):
            found.append(f".{node.attr}")
        elif isinstance(node, ast.Name) and node.id in FORBIDDEN_CALLS:
            found.append(node.id)
    return sorted(set(found))
```
